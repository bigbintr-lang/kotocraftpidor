/* ---------- Кнопка «Нет» убегает от мышки ---------- */ const no = document.getElementById('no'); const noSlot = document.getElementById('noSlot'); const texts = ['Нет 😿', 'Не надо~ 🥺', 'Ну пожалуйста!', 'Не лови меня 😭', 'Сенпай!!', 'Кавайно же!', 'Хихи, мимо~ 🎀']; let n = 0, fled = false; // выносим кнопку из карточки, чтобы fixed считался от окна document.body.appendChild(no); no.style.position = 'fixed'; no.style.zIndex = 50; function placeNo() { const r = noSlot.getBoundingClientRect(); no.style.transition = 'none'; no.style.left = r.left + 'px'; no.style.top = r.top + 'px'; } // пока кнопку не спугнули, она следует за своим местом в карточке (карточка плавает) (function follow() { if (!fled) placeNo(); requestAnimationFrame(follow); })(); function moveNo(x, y) { fled = true; no.style.transition = 'left .18s ease-out, top .18s ease-out'; const w = no.offsetWidth, h = no.offsetHeight, pad = 10; no.style.left = Math.max(pad, Math.min(innerWidth - w - pad, x)) + 'px'; no.style.top = Math.max(pad, Math.min(innerHeight - h - pad, y)) + 'px'; no.textContent = texts[++n % texts.length]; } function flee(px, py) { const r = no.getBoundingClientRect(); const cx = r.left + r.width / 2, cy = r.top + r.height / 2; let dx = cx - px, dy = cy - py; const dist = Math.hypot(dx, dy) || 1; const danger = 150; if (dist > danger) return; dx /= dist; dy /= dist; const step = 200 + Math.random() * 100; let nx = r.left + dx * step + (Math.random() - .5) * 80; let ny = r.top + dy * step + (Math.random() - .5) * 80; const cornered = nx < 10 || ny < 10 || nx > innerWidth - r.width - 10 || ny > innerHeight - r.height - 10; if (cornered && Math.random() < 0.55) { nx = Math.random() * (innerWidth - r.width - 20) + 10; ny = Math.random() * (innerHeight - r.height - 20) + 10; } moveNo(nx, ny); } addEventListener('pointermove', e => flee(e.clientX, e.clientY)); no.addEventListener('touchstart', e => { e.preventDefault(); const t = e.touches[0]; flee(t.clientX, t.clientY); }, { passive: false }); /* ---------- Если поймал «Нет» → грустный экран ---------- */ const sad = document.getElementById('sad'); no.addEventListener('click', () => { sad.classList.add('show'); for (let i = 0; i < 40; i++) { const t = document.createElement('div'); t.className = 'tear'; t.textContent = i % 3 ? '💧' : '😢'; t.style.left = Math.random() * 100 + 'vw'; t.style.animationDuration = 2 + Math.random() * 3 + 's'; t.style.animationDelay = -Math.random() * 4 + 's'; sad.appendChild(t); } }); document.getElementById('back').addEventListener('click', () => { sad.classList.remove('show'); sad.querySelectorAll('.tear').forEach(t => t.remove()); no.textContent = 'Нет 😿'; n = 0; fled = false; // кнопка вернётся на своё место в карточке });
